/**
 * Evidence, not decoration: where the last 60 sessions landed, and where today
 * sits relative to them.
 */
export function DistributionChart({ signal }: { signal: Signal }) {
  const [mounted, setMounted] = useState(false);
  useEffect(() => setMounted(true), []);

  const lower = signal.historicalMean - 2 * signal.historicalStdDev;
  const upper = signal.historicalMean + 2 * signal.historicalStdDev;
  const data = signal.distribution.map((b) => ({
    ...b,
    inRange: b.center >= lower && b.center <= upper,
  }));

  if (!mounted) return <div className="h-44 w-full rounded-lg bg-secondary/60" />;

  return (
    <figure className="mt-3">
      <div className="h-44 w-full">
        <ResponsiveContainer width="100%" height="100%">
          <BarChart data={data} margin={{ top: 18, right: 8, left: 8, bottom: 0 }} barCategoryGap={2}>
            <CartesianGrid vertical={false} stroke="var(--color-border)" />
            <XAxis
              dataKey="center"
              tickFormatter={(v: number) => `${v > 0 ? "+" : ""}${v.toFixed(1)}%`}
              tick={{ fontSize: 10, fill: "var(--color-muted-foreground)" }}
              stroke="var(--color-border)"
              interval={3}
            />
            <ReferenceLine
              x={data.reduce((best, d) =>
                Math.abs(d.center - signal.changePct) < Math.abs(best.center - signal.changePct)
                  ? d
                  : best,
              ).center}
              stroke={signal.direction === "up" ? "var(--color-up)" : "var(--color-down)"}
              strokeWidth={2}
              label={{
                value: "Today",
                position: "top",
                fontSize: 11,
                fill: signal.direction === "up" ? "var(--color-up)" : "var(--color-down)",
              }}
            />
            <Bar dataKey="count" radius={[2, 2, 0, 0]}>
              {data.map((d, i) => (
                <Cell
                  key={i}
                  fill={d.inRange ? "var(--color-chart-5)" : "var(--color-border-strong)"}
                />
              ))}
            </Bar>
          </BarChart>
        </ResponsiveContainer>
      </div>
      <figcaption className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-[11px] text-muted-foreground">
        <span className="flex items-center gap-1.5">
          <span className="size-2 rounded-sm bg-chart-5" aria-hidden /> Normal range (±2σ)
        </span>
        <span className="flex items-center gap-1.5">
          <span className="size-2 rounded-sm bg-border-strong" aria-hidden /> Tail observations
        </span>
        <span className="flex items-center gap-1.5">
          <span
            className={`h-3 w-0.5 ${signal.direction === "up" ? "bg-up" : "bg-down"}`}
            aria-hidden
          />
          Today's observation
        </span>
        <span>Distribution of daily returns, last {signal.lookbackDays} sessions</span>
      </figcaption>
    </figure>
  );
}

import type { FreshnessState } from "@/lib/signal-types";

const CONFIG: Record<
  FreshnessState,
  { label: string; dot: string; text: string; pulse: boolean }
> = {
  LIVE: { label: "Live", dot: "bg-live", text: "text-foreground", pulse: true },
  RECENT: { label: "Recent", dot: "bg-live/70", text: "text-foreground", pulse: false },
  DELAYED: { label: "Delayed", dot: "bg-delayed", text: "text-foreground", pulse: false },
  STALE: { label: "Stale", dot: "bg-stale", text: "text-foreground", pulse: false },
  UNAVAILABLE: { label: "Data unavailable", dot: "bg-muted-foreground", text: "text-foreground", pulse: false },
};

function detail(state: FreshnessState, secondsAgo: number) {
  switch (state) {
    case "LIVE":
    case "RECENT":
      return `Updated ${secondsAgo}s ago`;
    case "DELAYED":
      return "Provider is responding slowly";
    case "STALE":
      return "Data hasn't refreshed recently";
    case "UNAVAILABLE":
      return "Showing last known signals";
  }
}

export function FreshnessIndicator({
  state,
  secondsAgo,
  className,
}: {
  state: FreshnessState;
  secondsAgo: number;
  className?: string;
}) {
  const c = CONFIG[state];
  return (
    <div
      className={cn("flex items-center gap-2", className)}
      role="status"
      aria-live="polite"
    >
      <span className="relative flex size-2 shrink-0 items-center justify-center">
        <span className={cn("size-2 rounded-full", c.dot)} />
        {c.pulse && (
          <span className={cn("absolute size-2 animate-ping rounded-full opacity-60", c.dot)} />
        )}
      </span>
      <span className="leading-tight">
        <span
          className={cn(
            "block font-display text-[11px] font-semibold uppercase tracking-[0.08em]",
            c.text,
          )}
        >
          {c.label}
        </span>
        <span className="num block text-[11px] text-muted-foreground">
          {detail(state, secondsAgo)}
        </span>
      </span>
    </div>
  );
}

/** Intentionally simple: a chronological log, no extra analytics. */
export function HistoryList({ ticker }: { ticker: string }) {
  const items = historyFor(ticker);

  if (items.length === 0) {
    return (
      <p className="py-4 text-sm text-muted-foreground">
        No earlier signals recorded for {ticker} in the replay window.
      </p>
    );
  }

  return (
    <ul className="divide-y divide-border">
      {items.map((h) => (
        <li key={h.date} className="flex items-center justify-between gap-4 py-3">
          <div className="min-w-0">
            <p className="num text-sm font-medium">
              {new Date(h.date).toLocaleDateString("en-IN", {
                day: "2-digit",
                month: "short",
                year: "numeric",
              })}
            </p>
            <p className="text-xs text-muted-foreground">
              {SEVERITY_LABEL[h.severity]} ·{" "}
              {h.signalType === "stock_specific" ? "Stock-specific" : "Sector-wide"} ·{" "}
              {h.direction === "up" ? "Upside" : "Downside"}
            </p>
          </div>
          <span
            className={cn(
              "num font-display text-sm font-semibold",
              h.direction === "up" ? "text-up" : "text-down",
            )}
          >
            {formatSigma(h.zScore)}
          </span>
        </li>
      ))}
    </ul>
  );
}

  );

export function WatchlistPanel({ watchlist, onAdd, onRemove }: Props) {
  const [query, setQuery] = useState("");

  const results = useMemo(() => {
    if (query.trim().length < 1) return [];
    const q = query.toLowerCase();
    return SEARCH_UNIVERSE.filter(
      (s) =>
        (s.ticker.toLowerCase().includes(q) || s.company.toLowerCase().includes(q)) &&
        !watchlist.some((w) => w.ticker === s.ticker),
    ).slice(0, 5);
  }, [query, watchlist]);

  return (
    <section aria-labelledby="watchlist-heading" className="rounded-xl border border-border bg-surface shadow-card">
      <div className="flex items-center justify-between gap-3 border-b border-border px-4 py-3">
        <h2 id="watchlist-heading" className="eyebrow">
          Watchlist
        </h2>
        <span className="num text-xs text-muted-foreground">{watchlist.length} tracked</span>
      </div>

      <div className="border-b border-border px-4 py-3">
        <label className="relative block">
          <span className="sr-only">Search stocks to add</span>
          <Search
            className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
            aria-hidden
          />
          <Input
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            placeholder="Add a stock — e.g. TITAN"
            className="h-10 pl-9"
          />
        </label>
        {results.length > 0 && (
          <ul className="mt-2 divide-y divide-border overflow-hidden rounded-lg border border-border">
            {results.map((r) => (
              <li key={r.ticker} className="flex items-center justify-between gap-3 px-3 py-2">
                <div className="min-w-0">
                  <p className="text-sm font-medium">{r.ticker}</p>
                  <p className="truncate text-xs text-muted-foreground">
                    {r.company} · {r.sector}
                  </p>
                </div>
                <Button
                  size="sm"
                  variant="secondary"
                  onClick={() => {
                    onAdd(r.ticker);
                    setQuery("");
                  }}
                >
                  <Plus className="size-4" aria-hidden /> Add
                </Button>
              </li>
            ))}
          </ul>
        )}
      </div>

      <ul className="divide-y divide-border">
        {watchlist.map((item) => (
          <li key={item.ticker}>
            <Collapsible>
              <div className="group flex items-center gap-3 px-4 py-3 transition-colors hover:bg-secondary/60">
                <div className="min-w-0 flex-1">
                  <p className="text-sm font-medium">{item.ticker}</p>
                  <p className="truncate text-xs text-muted-foreground">
                    {item.company} · {item.sector}
                  </p>
                </div>
                <div className="text-right">
                  <p className="num text-sm font-medium">{formatINR(item.price)}</p>
                  <p
                    className={cn(
                      "num text-xs",
                      item.changePct > 0 ? "text-up" : item.changePct < 0 ? "text-down" : "text-muted-foreground",
                    )}
                  >
                    {formatPct(item.changePct)}
                  </p>
                </div>
                <CollapsibleTrigger asChild>
                  <Button size="icon" variant="ghost" aria-label={`View signal history for ${item.ticker}`}>
                    <History className="size-4" aria-hidden />
                  </Button>
                </CollapsibleTrigger>
                <Button
                  size="icon"
                  variant="ghost"
                  aria-label={`Remove ${item.ticker} from watchlist`}
                  onClick={() => onRemove(item.ticker)}
                >
                  <Trash2 className="size-4" aria-hidden />
                </Button>
              </div>
              <CollapsibleContent>
                <div className="border-t border-border bg-secondary/40 px-4 pb-2">
                  <p className="eyebrow pt-3">Signal history</p>
                  <HistoryList ticker={item.ticker} />
                </div>
              </CollapsibleContent>
            </Collapsible>
          </li>
        ))}
      </ul>

      {watchlist.length === 0 && (
        <p className="px-4 py-8 text-center text-sm text-muted-foreground">
          Your watchlist is empty. Search above to start tracking a stock.
        </p>
      )}
    </section>
  );
}

function ErrorComponent({ error, reset }: { error: Error; reset: () => void }) {
  console.error(error);
  const router = useRouter();
  useEffect(() => {
    reportLovableError(error, { boundary: "tanstack_root_error_component" });
  }, [error]);

  return (
    <div className="flex min-h-screen items-center justify-center bg-background px-4">
      <div className="max-w-md text-center">
        <h1 className="text-xl font-semibold tracking-tight text-foreground">
          This page didn't load
        </h1>
        <p className="mt-2 text-sm text-muted-foreground">
          Something went wrong on our end. You can try refreshing or head back home.
        </p>
        <div className="mt-6 flex flex-wrap justify-center gap-2">
          <button
            onClick={() => {
              router.invalidate();
              reset();
            }}
            className="inline-flex items-center justify-center rounded-md bg-primary px-4 py-2 text-sm font-medium text-primary-foreground transition-colors hover:bg-primary/90"
          >
            Try again
          </button>
          <a
            href="/"
            className="inline-flex items-center justify-center rounded-md border border-input bg-background px-4 py-2 text-sm font-medium text-foreground transition-colors hover:bg-accent"
          >
            Go home
          </a>
        </div>
      </div>
    </div>
  );
}
